## 二、核心模块详细设计

### 2.1 事件捕获模块 (EventCapture)

#### 设计目标
- 实时捕获键盘和鼠标事件
- 低延迟(< 50ms)
- 不影响用户正常操作

#### 技术实现

```go
// internal/engine/events/capture.go
package events

import (
    "github.com/robotn/gohook"
    "github.com/go-vgo/robotgo"
)

// EventCapture 事件捕获器
type EventCapture struct {
    keyboardChan chan KeyboardEvent
    mouseChan    chan MouseEvent
    isRunning    bool
    eventBuffer  []Event
}

// NewEventCapture 创建事件捕获器
func NewEventCapture() *EventCapture {
    return &EventCapture{
        keyboardChan: make(chan KeyboardEvent, 1000),
        mouseChan:    make(chan MouseEvent, 1000),
        isRunning:    false,
        eventBuffer:  make([]Event, 0, 10000),
    }
}

// Start 开始捕获事件
func (ec *EventCapture) Start() error {
    if ec.isRunning {
        return fmt.Errorf("capture already started")
    }

    ec.isRunning = true

    // 启动键盘监听
    go ec.startKeyboardHook()

    // 启动鼠标监听
    go ec.startMouseHook()

    // 启动事件处理
    go ec.processEvents()

    return nil
}

// startKeyboardHook 启动键盘钩子
func (ec *EventCapture) startKeyboardHook() {
    evChan := hook.Start()

    for ev := range evChan {
        if !ec.isRunning {
            break
        }

        switch ev.Kind {
        case hook.KeyDown:
            ec.keyboardChan <- KeyboardEvent{
                Type:      KeyDown,
                Keycode:   ev.Keychar,
                Timestamp: time.Now().UnixNano(),
            }
        case hook.KeyUp:
            ec.keyboardChan <- KeyboardEvent{
                Type:      KeyUp,
                Keycode:   ev.Keychar,
                Timestamp: time.Now().UnixNano(),
            }
        }
    }
}

// startMouseHook 启动鼠标监听
func (ec *EventCapture) startMouseHook() {
    for ec.isRunning {
        // 获取鼠标位置
        x, y := robotgo.GetMousePos()

        // 监听鼠标事件
        mEvent := robotgo.AddEvents("mouse", 10)

        select {
        case ev := <-mEvent:
            if ev.Kind == hook.MouseClick {
                ec.mouseChan <- MouseEvent{
                    Type:      MouseClick,
                    X:         x,
                    Y:         y,
                    Button:    ev.Button,
                    Timestamp: time.Now().UnixNano(),
                }
            } else if ev.Kind == hook.MouseMove {
                ec.mouseChan <- MouseEvent{
                    Type:      MouseMove,
                    X:         x,
                    Y:         y,
                    Timestamp: time.Now().UnixNano(),
                }
            }
        default:
            time.Sleep(10 * time.Millisecond)
        }
    }
}

// processEvents 处理事件流
func (ec *EventCapture) processEvents() {
    for ec.isRunning {
        select {
        case ke := <-ec.keyboardChan:
            ec.eventBuffer = append(ec.eventBuffer, ke)

        case me := <-ec.mouseChan:
            ec.eventBuffer = append(ec.eventBuffer, me)

        case <-time.After(100 * time.Millisecond):
            // 定期刷新缓冲区
        }
    }
}

// Stop 停止捕获
func (ec *EventCapture) Stop() {
    ec.isRunning = false
    hook.End()
}

// GetEvents 获取捕获的事件
func (ec *EventCapture) GetEvents() []Event {
    return ec.eventBuffer
}

// ClearEvents 清空事件缓冲
func (ec *EventCapture) ClearEvents() {
    ec.eventBuffer = make([]Event, 0, 10000)
}
```

### 2.2 脚本生成模块 (Generator)

#### 设计目标
- 将操作序列转换为可读的JavaScript代码
- 代码格式化良好
- 支持自定义模板

#### 技术实现

```go
// internal/engine/generator/generator.go
package generator

import (
    "fmt"
    "strings"
    "time"
    "macro_operation/internal/engine/events"
)

// Generator 脚本生成器
type Generator struct {
    template     string
    indentLevel  int
    addComments  bool
    optimizeCode bool
}

// NewGenerator 创建脚本生成器
func NewGenerator() *Generator {
    return &Generator{
        template:     defaultTemplate,
        indentLevel:  0,
        addComments:  true,
        optimizeCode: true,
    }
}

// defaultTemplate 默认脚本模板
const defaultTemplate = `// 宏名称: %s
// 录制时间: %s
// 操作数量: %d

function main() {
%s
}

// 执行宏
main();
`

// Generate 生成JavaScript脚本
func (g *Generator) Generate(name string, events []events.Event) (string, error) {
    if len(events) == 0 {
        return "", fmt.Errorf("no events to generate")
    }

    var sb strings.Builder

    // 生成操作代码
    for i, event := range events {
        code, err := g.generateEventCode(event)
        if err != nil {
            return "", err
        }

        sb.WriteString(code)

        // 添加延迟(除了最后一个操作)
        if i < len(events)-1 {
            nextEvent := events[i+1]
            delay := nextEvent.GetTimestamp() - event.GetTimestamp()
            delayMs := delay / 1_000_000 // 纳秒转毫秒

            if delayMs > 0 {
                sb.WriteString(g.indent(fmt.Sprintf("sleep(%d);\n", delayMs)))
            }
        }
    }

    // 应用模板
    script := fmt.Sprintf(g.template,
        name,
        time.Now().Format("2006-01-02 15:04:05"),
        len(events),
        sb.String(),
    )

    return script, nil
}

// generateEventCode 生成单个事件的代码
func (g *Generator) generateEventCode(event events.Event) (string, error) {
    switch e := event.(type) {
    case events.KeyboardEvent:
        return g.generateKeyboardCode(e)
    case events.MouseEvent:
        return g.generateMouseCode(e)
    default:
        return "", fmt.Errorf("unsupported event type: %v", event.GetType())
    }
}

// generateKeyboardCode 生成键盘事件代码
func (g *Generator) generateKeyboardCode(event events.KeyboardEvent) (string, error) {
    switch event.Type {
    case events.KeyDown:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 按下键盘: %c\n", event.Keycode)) +
                   g.indent(fmt.Sprintf("keyDown('%c');\n", event.Keycode)), nil
        }
        return g.indent(fmt.Sprintf("keyDown('%c');\n", event.Keycode)), nil

    case events.KeyUp:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 释放键盘: %c\n", event.Keycode)) +
                   g.indent(fmt.Sprintf("keyUp('%c');\n", event.Keycode)), nil
        }
        return g.indent(fmt.Sprintf("keyUp('%c');\n", event.Keycode)), nil

    default:
        return "", fmt.Errorf("unsupported keyboard event type: %v", event.Type)
    }
}

// generateMouseCode 生成鼠标事件代码
func (g *Generator) generateMouseCode(event events.MouseEvent) (string, error) {
    switch event.Type {
    case events.MouseMove:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 鼠标移动到 (%d, %d)\n", event.X, event.Y)) +
                   g.indent(fmt.Sprintf("mouseMove(%d, %d);\n", event.X, event.Y)), nil
        }
        return g.indent(fmt.Sprintf("mouseMove(%d, %d);\n", event.X, event.Y)), nil

    case events.MouseClick:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 鼠标点击 (%d, %d) 按钮: %s\n",
                event.X, event.Y, event.Button)) +
                   g.indent(fmt.Sprintf("mouseClick('%s');\n", event.Button)), nil
        }
        return g.indent(fmt.Sprintf("mouseClick('%s');\n", event.Button)), nil

    case events.MouseDrag:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 鼠标拖拽\n")), nil
        }
        return g.indent(fmt.Sprintf("mouseDrag(%d, %d);\n", event.X, event.Y)), nil

    case events.MouseScroll:
        if g.addComments {
            return g.indent(fmt.Sprintf("// 鼠标滚动: %d\n", event.Delta)) +
                   g.indent(fmt.Sprintf("mouseScroll(%d);\n", event.Delta)), nil
        }
        return g.indent(fmt.Sprintf("mouseScroll(%d);\n", event.Delta)), nil

    default:
        return "", fmt.Errorf("unsupported mouse event type: %v", event.Type)
    }
}

// indent 生成缩进
func (g *Generator) indent(code string) string {
    indent := strings.Repeat("  ", g.indentLevel)
    return indent + code
}
```

### 2.3 脚本执行模块 (Executor)

#### 设计目标
- 安全执行JavaScript脚本
- 提供丰富的API接口
- 支持调试和日志

#### 技术实现

```go
// internal/engine/executor/executor.go
package executor

import (
    "fmt"
    "github.com/dop251/goja"
    "macro_operation/pkg/bindings"
)

// Executor 脚本执行器
type Executor struct {
    vm           *goja.Runtime
    isRunning    bool
    stopChan     chan struct{}
    bindings     *bindings.APIBindings
}

// NewExecutor 创建执行器
func NewExecutor() *Executor {
    vm := goja.New()

    executor := &Executor{
        vm:        vm,
        isRunning: false,
        stopChan:  make(chan struct{}),
        bindings:  bindings.NewAPIBindings(vm),
    }

    // 注册API
    executor.registerAPIs()

    return executor
}

// registerAPIs 注册API函数
func (e *Executor) registerAPIs() {
    // 鼠标操作API
    e.vm.Set("mouseMove", e.mouseMove)
    e.vm.Set("mouseClick", e.mouseClick)
    e.vm.Set("mouseDrag", e.mouseDrag)
    e.vm.Set("mouseScroll", e.mouseScroll)

    // 键盘操作API
    e.vm.Set("keyDown", e.keyDown)
    e.vm.Set("keyUp", e.keyUp)
    e.vm.Set("keyType", e.keyType)
    e.vm.Set("keyTap", e.keyTap)

    // 系统操作API
    e.vm.Set("sleep", e.sleep)
    e.vm.Set("screenshot", e.screenshot)
    e.vm.Set("log", e.log)

    // 控制流API
    e.vm.Set("loop", e.loop)
}

// Execute 执行脚本
func (e *Executor) Execute(script string) error {
    if e.isRunning {
        return fmt.Errorf("executor already running")
    }

    e.isRunning = true
    defer func() {
        e.isRunning = false
    }()

    // 编译脚本
    program, err := goja.Compile("", script, true)
    if err != nil {
        return fmt.Errorf("compile error: %w", err)
    }

    // 执行脚本
    _, err = e.vm.RunProgram(program)
    if err != nil {
        return fmt.Errorf("runtime error: %w", err)
    }

    return nil
}

// Stop 停止执行
func (e *Executor) Stop() {
    if e.isRunning {
        close(e.stopChan)
        e.isRunning = false
    }
}

// API函数实现

func (e *Executor) mouseMove(x, y int64) error {
    return e.bindings.MouseMove(int(x), int(y))
}

func (e *Executor) mouseClick(button string) error {
    return e.bindings.MouseClick(button)
}

func (e *Executor) mouseDrag(x1, y1, x2, y2 int64) error {
    return e.bindings.MouseDrag(int(x1), int(y1), int(x2), int(y2))
}

func (e *Executor) mouseScroll(delta int64) error {
    return e.bindings.MouseScroll(int(delta))
}

func (e *Executor) keyDown(key string) error {
    return e.bindings.KeyDown(key)
}

func (e *Executor) keyUp(key string) error {
    return e.bindings.KeyUp(key)
}

func (e *Executor) keyType(text string) error {
    return e.bindings.KeyType(text)
}

func (e *Executor) keyTap(key string) error {
    return e.bindings.KeyTap(key)
}

func (e *Executor) sleep(ms int64) error {
    select {
    case <-time.After(time.Duration(ms) * time.Millisecond):
        return nil
    case <-e.stopChan:
        return fmt.Errorf("execution stopped")
    }
}

func (e *Executor) screenshot(path string) error {
    return e.bindings.Screenshot(path)
}

func (e *Executor) log(msg string) {
    fmt.Printf("[LOG] %s\n", msg)
}

func (e *Executor) loop(count int, callback goja.Callable) error {
    for i := 0; i < count; i++ {
        _, err := e.vm.RunProgram(goja.MustCompile("",
            fmt.Sprintf("(%d)(function(){})", i), true))
        if err != nil {
            return err
        }

        _, err = callback(goja.Undefined())
        if err != nil {
            return err
        }
    }
    return nil
}
```

### 2.4 宏管理模块 (MacroService)

#### 设计目标
- 宏的CRUD操作
- 宏文件存储和管理
- 宏导入导出

#### 技术实现

```go
// internal/service/macro_service.go
package service

import (
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "sync"
    "macro_operation/internal/model"
    "macro_operation/internal/repository"
)

// MacroService 宏管理服务
type MacroService struct {
    repo       repository.MacroRepository
    cache      map[string]*model.Macro
    cacheMutex sync.RWMutex
}

// NewMacroService 创建宏管理服务
func NewMacroService(repo repository.MacroRepository) *MacroService {
    service := &MacroService{
        repo:  repo,
        cache: make(map[string]*model.Macro),
    }

    // 加载所有宏到缓存
    service.loadCache()

    return service
}

// loadCache 加载宏缓存
func (s *MacroService) loadCache() error {
    macros, err := s.repo.ListAll()
    if err != nil {
        return err
    }

    s.cacheMutex.Lock()
    defer s.cacheMutex.Unlock()

    s.cache = make(map[string]*model.Macro)
    for _, macro := range macros {
        s.cache[macro.Name] = macro
    }

    return nil
}

// Save 保存宏
func (s *MacroService) Save(macro *model.Macro) error {
    // 设置时间戳
    macro.UpdatedAt = time.Now()
    if macro.CreatedAt.IsZero() {
        macro.CreatedAt = macro.UpdatedAt
    }

    // 保存到存储
    if err := s.repo.Save(macro); err != nil {
        return err
    }

    // 更新缓存
    s.cacheMutex.Lock()
    s.cache[macro.Name] = macro
    s.cacheMutex.Unlock()

    return nil
}

// Load 加载宏
func (s *MacroService) Load(name string) (*model.Macro, error) {
    s.cacheMutex.RLock()
    macro, ok := s.cache[name]
    s.cacheMutex.RUnlock()

    if !ok {
        return nil, fmt.Errorf("macro not found: %s", name)
    }

    return macro, nil
}

// Delete 删除宏
func (s *MacroService) Delete(name string) error {
    // 从存储删除
    if err := s.repo.Delete(name); err != nil {
        return err
    }

    // 从缓存删除
    s.cacheMutex.Lock()
    delete(s.cache, name)
    s.cacheMutex.Unlock()

    return nil
}

// List 列出所有宏
func (s *MacroService) List() ([]*model.Macro, error) {
    s.cacheMutex.RLock()
    defer s.cacheMutex.RUnlock()

    macros := make([]*model.Macro, 0, len(s.cache))
    for _, macro := range s.cache {
        macros = append(macros, macro)
    }

    return macros, nil
}

// Rename 重命名宏
func (s *MacroService) Rename(oldName, newName string) error {
    // 加载旧宏
    macro, err := s.Load(oldName)
    if err != nil {
        return err
    }

    // 检查新名称是否已存在
    if _, exists := s.cache[newName]; exists {
        return fmt.Errorf("macro already exists: %s", newName)
    }

    // 删除旧宏
    if err := s.Delete(oldName); err != nil {
        return err
    }

    // 保存新宏
    macro.Name = newName
    return s.Save(macro)
}

// Import 导入宏
func (s *MacroService) Import(filePath string) error {
    data, err := os.ReadFile(filePath)
    if err != nil {
        return err
    }

    var macro model.Macro
    if err := json.Unmarshal(data, &macro); err != nil {
        return err
    }

    return s.Save(&macro)
}

// Export 导出宏
func (s *MacroService) Export(name, exportPath string) error {
    macro, err := s.Load(name)
    if err != nil {
        return err
    }

    data, err := json.MarshalIndent(macro, "", "  ")
    if err != nil {
        return err
    }

    return os.WriteFile(exportPath, data, 0644)
}
```

#### 数据模型

```go
// internal/model/macro.go
package model

import "time"

// Macro 宏模型
type Macro struct {
    Meta       MetaInfo               `json:"meta"`
    Config     Config                 `json:"config"`
    Script     Script                 `json:"script"`
    Operations []map[string]interface{} `json:"operations"`
}

// MetaInfo 元信息
type MetaInfo struct {
    Name        string    `json:"name"`
```

```go
// internal/repository/macro_repo.go
package repository

import (
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "macro_operation/internal/model"
)

// MacroRepository 宏存储接口
type MacroRepository interface {
    Save(macro *model.Macro) error
    Load(name string) (*model.Macro, error)
    Delete(name string) error
    ListAll() ([]*model.Macro, error)
}

// FileMacroRepository 文件存储实现
type FileMacroRepository struct {
    macrosDir string
}

// NewFileMacroRepository 创建文件存储
func NewFileMacroRepository(macrosDir string) (*FileMacroRepository, error) {
    // 确保目录存在
    if err := os.MkdirAll(macrosDir, 0755); err != nil {
        return nil, err
    }

    return &FileMacroRepository{
        macrosDir: macrosDir,
    }, nil
}

// Save 保存宏
func (r *FileMacroRepository) Save(macro *model.Macro) error {
    filePath := r.getMacroPath(macro.Name)

    data, err := json.MarshalIndent(macro, "", "  ")
    if err != nil {
        return err
    }

    return os.WriteFile(filePath, data, 0644)
}

// Load 加载宏
func (r *FileMacroRepository) Load(name string) (*model.Macro, error) {
    filePath := r.getMacroPath(name)

    data, err := os.ReadFile(filePath)
    if err != nil {
        return nil, err
    }

    var macro model.Macro
    if err := json.Unmarshal(data, &macro); err != nil {
        return nil, err
    }

    return &macro, nil
}

// Delete 删除宏
func (r *FileMacroRepository) Delete(name string) error {
    filePath := r.getMacroPath(name)
    return os.Remove(filePath)
}

// ListAll 列出所有宏
func (r *FileMacroRepository) ListAll() ([]*model.Macro, error) {
    files, err := os.ReadDir(r.macrosDir)
    if err != nil {
        return nil, err
    }

    var macros []*model.Macro
    for _, file := range files {
        if file.IsDir() {
            continue
        }

        if filepath.Ext(file.Name()) != ".json" {
            continue
        }

        name := file.Name()[:len(file.Name())-5] // 去掉.json后缀
        macro, err := r.Load(name)
        if err != nil {
            continue
        }

        macros = append(macros, macro)
    }

    return macros, nil
}

// getMacroPath 获取宏文件路径
func (r *FileMacroRepository) getMacroPath(name string) string {
    return filepath.Join(r.macrosDir, name+".json")
}
```

```go
// configs/config.go
package configs

import (

// Config 配置
type Config struct {
    App      AppConfig      `mapstructure:"app"`

    Storage  StorageConfig  `mapstructure:"storage"`
}

// AppConfig 应用配置
type AppConfig struct {
    Name        string `mapstructure:"name"`
    Version     string `mapstructure:"version"`
    Debug       bool   `mapstructure:"debug"`
    LogPath     string `mapstructure:"log_path"`
}

// RecordConfig 录制配置
type RecordConfig struct {
    MaxDuration     int  `mapstructure:"max_duration"`     // 最大录制时长(秒)
    EnableScreenshot bool `mapstructure:"enable_screenshot"` // 启用截图
    FilterMouseMove  bool `mapstructure:"filter_mouse_move"` // 过滤鼠标移动事件
}

// PlaybackConfig 回放配置
type PlaybackConfig struct {
    DefaultSpeed    float64 `mapstructure:"default_speed"`    // 默认播放速度
    EnableSound     bool    `mapstructure:"enable_sound"`     // 启用声音提示
    StopOnError     bool    `mapstructure:"stop_on_error"`    // 错误时停止
}

// StorageConfig 存储配置
type StorageConfig struct {
    MacrosDir      string `mapstructure:"macros_dir"`       // 宏文件目录
    BackupEnabled  bool   `mapstructure:"backup_enabled"`   // 启用备份
    BackupDir      string `mapstructure:"backup_dir"`       // 备份目录
    MaxBackupCount int    `mapstructure:"max_backup_count"` // 最大备份数
}

// Load 加载配置
func Load(configPath string) (*Config, error) {
    v := viper.New()

    // 设置配置文件
    v.SetConfigFile(configPath)
    v.SetConfigType("yaml")

    // 读取环境变量
    v.AutomaticEnv()

    // 读取配置文件
    if err := v.ReadInConfig(); err != nil {
        return nil, err
    }

    var config Config
    if err := v.Unmarshal(&config); err != nil {
        return nil, err
    }

    // 设置默认值
    setDefaults(&config)

    // 创建必要的目录
    createDirectories(&config)

    return &config, nil
}

// setDefaults 设置默认值
func setDefaults(config *Config) {
    if config.App.Name == "" {
        config.App.Name = "Macro Recorder"
    }
    if config.App.Version == "" {
        config.App.Version = "1.0.0"
    }
    if config.Record.MaxDuration == 0 {
        config.Record.MaxDuration = 3600 // 1小时
    }
    if config.Playback.DefaultSpeed == 0 {
        config.Playback.DefaultSpeed = 1.0
    }
    if config.Storage.MacrosDir == "" {
        config.Storage.MacrosDir = "./macros"
    }
}

// createDirectories 创建必要的目录
func createDirectories(config *Config) {
    dirs := []string{
        config.Storage.MacrosDir,
        config.App.LogPath,
        config.Storage.BackupDir,
    }

    for _, dir := range dirs {
        if dir != "" {
            os.MkdirAll(dir, 0755)
        }
    }
}
```
